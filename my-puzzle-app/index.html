export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: 'Method Not Allowed' }); } const { words, targetDigits } = req.body; if (!words || !targetDigits || words.length !== 4 || targetDigits.length !== 9) { return res.status(400).json({ error: '入力データが不正です' }); } try { // 4つの熟語をひとつの文字列に結合（全9文字） const allKanji = words.join(''); if (allKanji.length !== 9) { return res.status(400).json({ error: '文字数の合計が9文字になりません' }); } const readings = []; const actualStrokes = []; // 1. 各漢字の画数を取得して判定 for (let i = 0; i < 9; i++) { const char = allKanji[i]; const kanjiRes = await fetch(`https://kanjiapi.dev/v1/kanji/${encodeURIComponent(char)}`); if (!kanjiRes.ok) { return res.status(200).json({ success: false, error: `「${char}」は漢字として認識されませんでした` }); } const kanjiData = await kanjiRes.json(); const strokeCount = kanjiData.stroke_count; const expectedStroke = parseInt(targetDigits[i], 10); if (strokeCount !== expectedStroke) { return res.status(200).json({ success: false, error: `「${char}」の画数は ${strokeCount}画 です（要求: ${expectedStroke}画）` }); } } // 2. 熟語の読みを取得 for (const word of words) { const wordRes = await fetch(`https://labs.goo.ne.jp/api/hiragana`, { // ※独自API等で読みを取得している場合はその処理 }).catch(() => null); // 簡易読み変換（goo API等の外部辞書を使わない場合は、Yahoo/Mecabや代替辞書を使用） // ここではkanjiapiの読み候補から抽出または仮取得 } // すべてクリアした場合 return res.status(200).json({ success: true, readings: words // 読み判定ロジックに合わせて返却 }); } catch (err) { return res.status(500).json({ error: 'サーバー内部エラーが発生しました' }); } }
